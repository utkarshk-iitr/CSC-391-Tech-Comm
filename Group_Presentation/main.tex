\begin{frame}{\texttt{cat Related\_Work.txt}}

\textit{\large\textbf{Towards Measuring Supply Chain Attacks on Package Managers for Interpreted Languages}} \hfill --- Ruian Duan et al., NDSS 2021 \cite{duan2021}

{What problem does the paper address?}
\begin{itemize}
    \item Modern software heavily depends on third-party packages from registries like PyPI.
    \item These ecosystems assume implicit trust, which attackers exploit to inject malware.
\end{itemize}

{What does the paper aim to do?}
\begin{itemize}
    \item Provide systematic comparison of security weaknesses in package ecosystems.
    \item Identify {broken trust assumptions} and structural vulnerabilities enabling attacks.
    \item Build an automated pipeline (MALOSS) to {detect malicious packages at scale}.
\end{itemize}

\textit{Move from reactive, case-by-case fixes to a scalable, data-driven analysis of supply chain threats.}
\vspace{0.5em}
\hfill {\scriptsize Source: Duan et al.\cite{duan2021}, NDSS 2021 (Introduction, Methodology)}

\end{frame}

\begin{frame}{\texttt{cat Package\_Manager\_Ecosystem.txt}}
    \begin{center}
        \includegraphics[width=1\linewidth]{fig_1.png}
    {\scriptsize Source: Duan et al.\cite{duan2021}, NDSS 2021, Fig. 1.}
    \end{center}
    \vspace{-0.4em}
    \textbf{Flow \& Security Insight}
    \vspace{-0.3em}
\begin{itemize}
    \item Packages flow: developers $\rightarrow$ registries $\rightarrow$ applications $\rightarrow$ users.
    \item Attacks exploit upstream components such as packages, where trust is assumed.
    \item Impact spreads downstream across the ecosystem, affecting applications and users.
\end{itemize}
\end{frame}

\begin{frame}{\texttt{./MALOSS\_Pipeline.sh}}
    \begin{columns}[T]

    \begin{column}{0.4\textwidth}
    \vspace{1.5em}
    \centering
    \textbf{MAL}ware detection for \textbf{O}pen \textbf{S}ource \textbf{S}oftware
    \end{column}
    
    \begin{column}{0.6\textwidth}
    \centering
    \vspace{-0.5em}
    \includegraphics[width=\linewidth]{fig_2.png}\\[-1em]
    {\tiny MALOSS Pipeline, Source: Duan et al.\cite{duan2021}, NDSS 2021, Fig. 4.}
    \end{column}
    \end{columns}
    \vspace{-1.5em}
    \begin{enumerate}
        \item \textbf{Metadata analysis:}
        \begin{itemize}
            \item Flags suspicious packages using authors, releases, dependencies, and embedded bins.
            \item Helps detect typosquatting, author mismatch, and links to known malware.
        \end{itemize}

        \item \textbf{Static analysis:}
        \begin{itemize}
            \item Scans source code for risky APIs: network, filesystem, process, and code generation.
            \item Detects suspicious flows such as \textit{secret file $\rightarrow$ network exfiltration}.
        \end{itemize}

        \item \textbf{Dynamic analysis:}
        \begin{itemize}
            \item Runs packages in Docker sandboxes during install, import, and function execution.
            \item Traces DNS/IP activity, file access, and process creation using Sysdig.
        \end{itemize}

        \item \textbf{Human-in-the-loop verification:}
        \begin{itemize}
            \item Manual triage verifies flagged packages, confirms malware, and refines heuristic rules.
            \item Reduces false positives and improves the accuracy of the pipeline iteratively.
        \end{itemize}
    \end{enumerate}
    \vspace{-0.8em}
    {\scriptsize Source: Duan et al.\cite{duan2021}, NDSS 2021, Sec. III-B and Table III.}
\end{frame}

\begin{frame}{\texttt{head Key\_Findings.txt}}


\begin{columns}[T]
    \begin{column}{0.6\textwidth}
\textbf{Key findings from the paper}
        \begin{itemize}
            \item MALOSS analyzed \textbf{over 1 million packages} across PyPI, Npm, and RubyGems. 
            \item It identified \textbf{339 new malicious packages}; \textbf{278} were confirmed and removed by maintainers.
            \item \textbf{Three} removed packages had \textbf{more than 100K downloads} and received CVEs.
            \item \textbf{Typosquatting} and \textbf{account compromise} were the most common attack vectors.
            \item About \textbf{20\%} of malware stayed online for \textbf{over 400 days}, showing slow detection and removal.
        \end{itemize}

        {\scriptsize Source: Duan et al.\cite{duan2021}, NDSS 2021, Abstract and Sec. IV.}
    \end{column}

\begin{frame}{\texttt{more Mitigation\_Limitations.txt}}

\small
\begin{block}{Actionable mitigations}
\begin{itemize}
    \item \textbf{Registry maintainers:} enforce MFA, automated vetting, typo detection, and fast advisories.
    \item \textbf{Package maintainers:} secure infrastructure, restrict access, and use safe workflows.
    \item \textbf{Developers:} pin dependencies, review install scripts, and treat updates as security events.
    \item \textbf{Ecosystem response:} notify dependents quickly and control releases of popular packages.
\end{itemize}
\end{block}

\begin{block}{Limitations}
\begin{itemize}
    \item Static/dynamic analysis may produce false positives and miss hidden behaviors.
    \item Dynamic analysis depends on execution paths, trigger-based malware can evade detection.
    \item Large dependency graphs make complete analysis difficult.
    \item Manual verification is required, limits full automation.
    \item Attackers can evade detection using obfuscation or delayed execution.
\end{itemize}
\end{block}

\vspace{-0.5em}
{\scriptsize Source: Duan et al.\cite{duan2021}, NDSS 2021, Sec. III-B, Sec. IV and Sec. V.}

\end{frame}
