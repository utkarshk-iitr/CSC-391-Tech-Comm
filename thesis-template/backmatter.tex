\section{Example Dependency Policy Script}

\begin{lstlisting}[language=Python,caption={Illustrative policy checks for dependency metadata.},label={lst:policy-check}]
import json

def evaluate_dependency(meta):
	risk = 0

	if not meta.get("signed_release", False):
		risk += 3
	if meta.get("new_maintainer", False):
		risk += 2
	if meta.get("suspicious_install_script", False):
		risk += 4
	if meta.get("typosquat_score", 0) > 0.8:
		risk += 3

	if risk >= 7:
		return "quarantine"
	if risk >= 4:
		return "manual_review"
	return "allow"


if __name__ == "__main__":
	sample = {
		"signed_release": False,
		"new_maintainer": True,
		"suspicious_install_script": True,
		"typosquat_score": 0.1,
	}
	print(evaluate_dependency(sample))
\end{lstlisting}

set -e

echo "Running dependency integrity checks..."
python scripts/policy_check.py

echo "Running static source checks..."
./scripts/static_scan.sh

./scripts/enforce_threshold.sh
